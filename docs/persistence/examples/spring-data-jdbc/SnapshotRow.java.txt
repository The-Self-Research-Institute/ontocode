package self.research.ontology.reasoner.repository.springjdbc;

import lombok.AccessLevel;
import lombok.Getter;
import lombok.NoArgsConstructor;
import lombok.Setter;
import org.springframework.data.annotation.Id;
import org.springframework.data.annotation.Transient;
import org.springframework.data.domain.Persistable;
import org.springframework.data.relational.core.mapping.Table;

import java.time.Instant;

@Getter
@Setter
@NoArgsConstructor
@Table("reasoning_snapshot")
public class SnapshotRow implements Persistable<String> {

    @Id
    private String id;
    private String projectId;
    private long revision;
    private long tripleCount;
    private String reasoner;
    private String rulesFingerprint;
    private String resultJson;
    private Instant createdAt;
    private Instant updatedAt;

    @Transient
    @Getter(AccessLevel.NONE)
    @Setter(AccessLevel.NONE)
    private boolean newRow;

    @Override
    public boolean isNew() {
        return newRow;
    }

    public void markNew(boolean value) {
        this.newRow = value;
    }
}
